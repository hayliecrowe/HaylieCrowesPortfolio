                function Particle(x, y) {

                    this.x = x;

                    this.y = y;

                    this.size = Math.random() * 0.4 + 0.2; 

                    this.timeOffset = Math.random() * Math.PI * 2; 

                    

                    this.getParticleColor = function() {

                         const isLightMode = body.classList.contains('light-mode');

                         // Gold tint, very faint opacity (0.05 to 0.15)

                         const hue = isLightMode ? 40 : 45; 

                         const saturation = isLightMode ? 50 : 60;

                         const lightness = isLightMode ? 60 : 75;

                         const opacity = Math.random() * 0.1 + 0.05;


                         return 'hsla(' + hue + ', ' + saturation + '%, ' + lightness + '%, ' + opacity + ')'; 

                    };

                    this.color = this.getParticleColor();

                }


                Particle.prototype.update = function(time) {

                    this.x += Math.sin(time * 0.0001 + this.timeOffset) * 0.2; 

                    this.y += Math.cos(time * 0.00015 + this.timeOffset) * 0.1; 


                    if (this.x > canvas.width * 1.5 || this.x < -canvas.width * 0.5 ||

                        this.y > canvas.height * 1.5 || this.y < -canvas.height * 0.5) {

                        

                        if (Math.random() < 0.5) {

                            this.x = Math.random() < 0.5 ? -10 : canvas.width + 10;

                            this.y = Math.random() * canvas.height;

                        } else {

                            this.x = Math.random() * canvas.width;

                            this.y = Math.random() < 0.5 ? -10 : canvas.height + 10;

                        }

                        this.color = this.getParticleColor(); 

                    }

                };


                Particle.prototype.draw = function() {

                    ctx.fillStyle = this.color;

                    ctx.beginPath();

                    ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);

                    ctx.fill();

                };

                    }

                }

                

                function animateParticles(timestamp) {

                    if (!lastTime) lastTime = timestamp; 


                    const bgColor = getComputedStyle(document.body).getPropertyValue('--c-bg-primary').trim();

                    ctx.fillStyle = bgColor; 

                    ctx.fillRect(0, 0, canvas.width, canvas.height);

                    

                    for (let i = 0; i < particles.length; i++) {

                        particles[i].update(timestamp); 

                        particles[i].draw();

                    }

                    

                    requestAnimationFrame(animateParticles);

                }

                

                animateParticles();

            }
